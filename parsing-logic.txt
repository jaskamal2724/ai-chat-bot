function splitOnNumber(str) {
  
  const match = str.match(/(\d+)\.$/);
  if (match) {
    const number = match[0]; // e.g., "1."
    const text = str.slice(0, -number.length); 
    return [text.trim(), number];
  }
  return [str]; 
}
